February 2010

    "Documents.TestHelpers.Files.test.msg")) {
  const int bufferLength = 256;
  var buffer = new Byte\[bufferLength\];
  if (stream != null) {
    int bytesRead = stream.Read(buffer, 0, bufferLength);
